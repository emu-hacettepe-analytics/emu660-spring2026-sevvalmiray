compute_stats <- function(x) {
  if (!is.numeric(x)) {
    stop("Error: It should take a numeric vector!")
  }
  
  stats_list <- list(
    mean = mean(x),
    median = median(x),
    variance = var(x),
    iqr = IQR(x),
    min = min(x),
    max = max(x)
  )
  return(stats_list)
}
